on_actions = {
    on_startup = {
        effect = {
            # VNR enable flag, so that other mods will load oob designed for VNR 
			set_global_flag = vnr_enabled
            # VNR language
            set_global_flag = navy_rework_english
            # set_global_flag = navy_rework_chinese
			
            # show the first panel of intro splash
			every_country = {
                limit = { is_ai = no }
                # show welcome splash
			    set_country_flag = navy_rework_splash_show
				set_variable = { navy_rework_welcome_splash_variable = 1 }
			}

            set_variable = { vnr_ai_difficulty = 0 }
			set_variable = { vnr_ai_production = 0 }

            # major naval powers array setup
            add_to_array = { global.vnr_naval_powers = EQS }
            add_to_array = { global.vnr_naval_powers = JER }
            add_to_array = { global.vnr_naval_powers = CHN }
            add_to_array = { global.vnr_naval_powers = WNG }
            add_to_array = { global.vnr_naval_powers = WNG }
            add_to_array = { global.vnr_naval_powers = STG }
            add_to_array = { global.vnr_naval_powers = HIP }
			

            # naval treaty violator
            set_global_flag = { flag = MTG_naval_treaty_violator value = 0 }
        }
    }
}
